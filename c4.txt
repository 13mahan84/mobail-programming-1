<ion-app>

  <ion-header>
    <ion-toolbar color="primary">
      <ion-title>مدیریت دانشجویان</ion-title>
    </ion-toolbar>
  </ion-header>

  <ion-content class="ion-padding">

    <ion-item>
      <ion-input
        id="studentName"
        label="نام دانشجو"
        placeholder="نام را وارد کنید">
      </ion-input>
    </ion-item>

    <ion-button expand="block" onclick="addStudent()">
      ثبت دانشجو
    </ion-button>

    <ion-list id="studentList">
    </ion-list>

  </ion-content>

</ion-app>

<script>
  function addStudent() {

    const input = document.getElementById("studentName");
    const name = input.value.trim();

    if (name === "") {
      alert("لطفاً نام دانشجو را وارد کنید");
      return;
    }

    const item = document.createElement("ion-item");
    item.textContent = name;

    document.getElementById("studentList").appendChild(item);

    input.value = "";
  }
</script>
